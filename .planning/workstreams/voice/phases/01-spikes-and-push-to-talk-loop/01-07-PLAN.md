<!-- From @blip/protocol. Derive every type from PayloadOf; never redeclare payload shapes. -->
buildFrame(type, payload) -> { v:1, id, type, ts, payload } (throws on an invalid payload)
parseFrame(raw: string) -> { ok: true, message } | { ok: false, code, message }
PayloadOf<"utterance"> = { utteranceId: string; text: string (<= MAX_UTTERANCE_CHARS = 500); source: "voice" | "typed"; sttConfidence?: number; alternatives?: string[] }
PayloadOf<"ptt"> = { state: "start" | "end" }
PayloadOf<"speak"> = { speakId: string; text: string; category: "answer" | "filler" | "readback" | "confirm-prompt" | "error" | "result"; interrupt: boolean; spell?: boolean }
PayloadOf<"speak.done"> = { speakId: string; interrupted: boolean }
PayloadOf<"earcon">["name"] = "listening" | "thinking" | "done" | "blocked" | "error"
PayloadOf<"state"> = { dialogue, page | null, focusFieldId?, busy: boolean, extensionConnected: boolean, brain: "claude" | "fake" }
Hello = { role: "voice", clientVersion: string, protocolVersion: 1 }; welcome = { sessionId, serverVersion, brain }
CLOSE_CODES.replaced = 4004; WS_PATH = "/ws"; newId()
</interfaces>

| server message to DOM | Replies derive from page text and must not become markup |
